import sys
input = sys.stdin.readline

MOD = 10**9 + 7

def modexp(a, e):
    res = 1
    while e:
        if e & 1:
            res = (res * a) % MOD
        a = (a * a) % MOD
        e >>= 1
    return res

def solve():
    x, y = map(int, input().split())
    
    if y % x != 0:
        print(0)
        return
    
    k = y // x
    
    if k == 1:
        print(1)
        return
    
    # compute divisors of k
    divs = []
    i = 1
    while i * i <= k:
        if k % i == 0:
            divs.append(i)
            if i * i != k:
                divs.append(k // i)
        i += 1
    
    # Möbius via factorization per divisor
    def mobius(n):
        cnt = 0
        i = 2
        while i * i <= n:
            if n % i == 0:
                if (n // i) % i == 0:
                    return 0
                cnt += 1
                n //= i
            else:
                i += 1
        if n > 1:
            cnt += 1
        return -1 if cnt % 2 else 1
    
    ans = 0
    for d in divs:
        mu = mobius(d)
        if mu == 0:
            continue
        val = modexp(2, k // d - 1)
        ans = (ans + mu * val) % MOD
    
    print(ans % MOD)

    solve()
