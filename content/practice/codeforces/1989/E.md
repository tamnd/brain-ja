## Python ソリューション```python
import sys
input = sys.stdin.readline

MOD = 998244353

def solve():
    n, k = map(int, input().split())
    if k == 1:
        return "1"

    # precompute binomial coefficients C(n-1, i)
    m = n - 1
    C = [0] * (k)
    C[0] = 1
    for i in range(1, k):
        C[i] = C[i - 1] * (m - i + 1) // i

    total = 1 << m

    bad = 0
    for i in range(k - 1):
        bad += C[i]

    return str((total - bad) % MOD)

t = int(input())
out = []
for _ in range(t):
    out.append(solve())

print("\n".join(out))

## テストケース```python
import sys, io

MOD = 998244353

def run(inp: str) -> str:
    sys.stdin = io.StringIO(inp)
    input = sys.stdin.readline

    def solve():
        n, k = map(int, input().split())
        if k == 1:
            return "1"
        m = n - 1

        C = [0] * (k)
        C[0] = 1
        for i in range(1, k):
            C[i] = C[i - 1] * (m - i + 1) // i

    t = int(input())
    return "\n".join(solve() for _ in range(t))
