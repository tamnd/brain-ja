## Python ソリューション```python
import sys
input = sys.stdin.readline

def can_form_snowflake(n):
    if n == 1:
        return False
    k = 2
    while k * k <= n:
        total = 1
        power = 1
        while total < n:
            power *= k
            total += power
        if total == n:
            return True
        k += 1
    return False

t = int(input())
for _ in range(t):
    n = int(input())
    print("YES" if can_form_snowflake(n) else "NO")
