## Python ソリューション```python
import sys
input = sys.stdin.readline

def geom_sum(k, h, limit):
    total = 1
    cur = 1
    for _ in range(h):
        cur *= k
        total += cur
        if total > limit:
            return limit + 1
    return total

def ok(n):
    if n < 3:
        return False

    # maximum height needed is small because 2^60 > 1e18
    for h in range(1, 61):
        lo, hi = 2, n

        while lo <= hi:
            mid = (lo + hi) // 2
            val = geom_sum(mid, h, n)

            if val == n:
                return True
            if val < n:
                lo = mid + 1
            else:
                hi = mid - 1

    return False

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        print("YES" if ok(n) else "NO")

if __name__ == "__main__":
    solve()
