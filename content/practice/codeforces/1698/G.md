## テストケース```python
import sys, io

def run(inp: str) -> str:
    sys.stdin = io.StringIO(inp)
    s = input().strip()
    if '1' not in s:
        return "-1"
    l = s.index('1')
    r = len(s) - 1 - s[::-1].index('1')
    p = 1
    q = 1 + (r - l + 1)
    return f"{p} {q}"

# provided samples
