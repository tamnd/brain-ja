 4. Строим четыре уравнения седьмого столбца, если седьмой столбец сохранился. Сначала формируем множество элементов`S`、затем четыре диагональные группы。 Каждый элемент представлен парой`(row, column)`、поэтому коэффициент неизвестной переменной можно получить простым переключением бита в маске.

 Использование маски особенно удобно, потому что все коэффициенты равны только нулю или единице. Если один и тот же элемент встретится дважды в XOR, его коэффициент должен стать нулём, поскольку`x XOR x = 0`。 
5. Полученная система квадратная。 При двух потерянных исходных столбцах она имеет размер`8 × 8`、 при потере исходного столбца и контрольного столбца размер равен`4 × 4`。 Обращаем эту матрицу один раз методом Гаусса над`GF(2)`。 
6. Лждой группы из четырёх чисел вычисляем правые части всех доступных контрольных уравнений. Правую часть получаем XOR-ом сохранённого контрольного значения с теми известными исходными значениями, которые входят в сответствующее уравнение.
 7. Получаем неизвестные значения умножением обратной матрицы на вектор правых частей. В поле`GF(2)`это означает обычный XOR выбранных компонент вектора. Поскольку обратная матрица заранее известна, здесь больге нет Гаусса.
 8. 最高のパフォーマンスを発揮します。 После этого, если среди потерянных столбцов есть зестой или седьмой, вычисляем его напрямую по уже Сосстановленным исходным столбцам。 
9. Выводим два массива в порядке`e1`、`e2`。 Каждый массив содержит все`n`чисел、включая значения из разных`4 × 5`そうです。 

Почему это работает: для каждой группы из четырёх строк мы строим ровно ту систему XOR-уравнений, которая была Подировании です。 Все известные слагаемые переносятся в правую часть, а неизвестные остаются слева. Матрица коэффициентов имеет единственное резение для каждого допустимого заблона двух потерь, поскольку по условию код восстанавливает любые два столбца. Предвычисленная обратная матрица возвращает именно это единственное резение. После восстановления исходных столбцов контрольные столбцы вычисляются непосредственно из исходной формулы, Поэтому итоговые значения совпадают с теми, которые были потеряны.

 ## Python ソリューション```python
import sys
input = sys.stdin.readline

def build_terms():
    """Return the list of data cells used by every row of columns 6 and 7."""
    terms6 = []
    for i in range(4):
        terms6.append([(i, c) for c in range(5)])

    s_cells = [(0, 4), (1, 3), (2, 2), (3, 1)]

    terms7 = []
    for i in range(4):
        cells = s_cells[:]

        for k in range(4):
            c = (i - k) % 5
            cells.append((k, c))

        # XOR is addition modulo 2, so duplicate cells cancel.
        parity = {}
        for cell in cells:
            parity[cell] = parity.get(cell, 0) ^ 1

        terms7.append([cell for cell, coef in parity.items() if coef])

    return terms6, terms7

def invert_binary_matrix(matrix):
    """Invert a square matrix whose entries are 0/1.

    Rows are represented by bit masks.
    The returned value contains, for every result variable,
    a bit mask of RHS equations that must be XORed.
    """
    m = len(matrix)
    rows = []

    for i, mask in enumerate(matrix):
        rows.append(mask | (1 << (m + i)))

    for col in range(m):
        pivot = -1
        for row in range(col, m):
            if (rows[row] >> col) & 1:
                pivot = row
                break

        if pivot == -1:
            raise ValueError("Singular matrix")

        rows[col], rows[pivot] = rows[pivot], rows[col]

        for row in range(m):
            if row != col and ((rows[row] >> col) & 1):
                rows[row] ^= rows[col]

    rhs_mask = (1 << m) - 1
    return [(rows[i] >> m) & rhs_mask for i in range(m)]

def solve():
    n, e1, e2 = map(int, input().split())
    lost = {e1, e2}

    full = [None] * 8
    for c in range(1, 8):
        if c not in lost:
            full[c] = list(map(int, input().split()))

    terms6, terms7 = build_terms()
    parity_terms = {6: terms6, 7: terms7}

    unknown = []
    for c in range(1, 6):
        if c in lost:
            for r in range(4):
                unknown.append((r, c - 1))

    position = {cell: i for i, cell in enumerate(unknown)}
    m = len(unknown)

    equations = []

    for p in (6, 7):
        if p in lost:
            continue

        for i in range(4):
            coeff = 0
            known_terms = []

            for cell in parity_terms[p][i]:
                if cell in position:
                    coeff ^= 1 << position[cell]
                else:
                    known_terms.append(cell)

            equations.append((p, i, coeff, known_terms))

    if m:
        matrix = [eq[2] for eq in equations]
        inverse = invert_binary_matrix(matrix)

        # Convert each inverse row to a compact list of RHS indices.
        combinations = []
        for mask in inverse:
            indices = []
            while mask:
                bit = mask & -mask
                indices.append(bit.bit_length() - 1)
                mask ^= bit
            combinations.append(indices)

        for start in range(0, n, 4):
            rhs = []

            for p, i, _, known_terms in equations:
                value = full[p][start + i]

                for r, c in known_terms:
                    value ^= full[c + 1][start + r]

                rhs.append(value)

            answer = [0] * m

            for var in range(m):
                value = 0
                for eq_index in combinations[var]:
                    value ^= rhs[eq_index]
                answer[var] = value

            for var, (r, c) in enumerate(unknown):
                full[c + 1][start + r] = answer[var]

    # At this point all five original data columns are known.
    # Rebuild every missing parity column directly.
    if 6 in lost:
        full[6] = [0] * n
        for start in range(0, n, 4):
            for r in range(4):
                value = 0
                for c in range(1, 6):
                    value ^= full[c][start + r]
                full[6][start + r] = value

    if 7 in lost:
        full[7] = [0] * n
        for start in range(0, n, 4):
            for r in range(4):
                value = 0
                for rr, c in terms7[r]:
                    value ^= full[c + 1][start + rr]
                full[7][start + r] = value

    sys.stdout.write(
        " ".join(map(str, full[e1])) + "\n" +
        " ".join(map(str, full[e2])) + "\n"
    )

if __name__ == "__main__":
    solve()

 При`n = 2 * 10^5`число групп равно 50000. Матрица имеет размер не более`8 × 8`、поэтому её построение и обращение занимает пренебрежимо малое время по сравнению с чтением и обработкой миллиона входных чисел。 Основная работа линейна по размеру входа и укладывается в ограничение памяти 256 МБ.

def build_terms_test():
    terms6 = []
    for i in range(4):
        terms6.append([(i, c) for c in range(5)])

    s_cells = [(0, 4), (1, 3), (2, 2), (3, 1)]
    terms7 = []

    for i in range(4):
        parity = {}
        for cell in s_cells:
            parity[cell] = parity.get(cell, 0) ^ 1

        for k in range(4):
            cell = (k, (i - k) % 5)
            parity[cell] = parity.get(cell, 0) ^ 1

        terms7.append([cell for cell, coef in parity.items() if coef])

    return terms6, terms7

def encode(data):
    n = len(data[0])
    terms6, terms7 = build_terms_test()

    cols = [None] + [list(col) for col in data] + [None, None]

    cols[6] = [0] * n
    cols[7] = [0] * n

    for start in range(0, n, 4):
        for r in range(4):
            value = 0
            for c in range(1, 6):
                value ^= cols[c][start + r]
            cols[6][start + r] = value

        for r in range(4):
            value = 0
            for rr, c in terms7[r]:
                value ^= cols[c + 1][start + rr]
