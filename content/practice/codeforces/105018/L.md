 ## 問題の理解

変換が呼び出されます$m$-すべての選択肢の中でこの量が最大になる場合は良好です$a, b, c, d \in [0, m-1]$そして、そのような変換が最大値に達する回数を数えるよう求められます。

## アプローチ

## Python ソリューション```python
import sys
input = sys.stdin.readline

MOD = 10**9 + 7
MAXM = 10**6

spf = list(range(MAXM + 1))
for i in range(2, int(MAXM ** 0.5) + 1):
    if spf[i] == i:
        for j in range(i * i, MAXM + 1, i):
            if spf[j] == j:
                spf[j] = i

def factorize(x):
    primes = set()
    while x > 1:
        p = spf[x]
        primes.add(p)
        x //= p
    return primes

inv_cache = {}

def modinv(x):
    if x in inv_cache:
        return inv_cache[x]
    inv_cache[x] = pow(x, MOD - 2, MOD)
    return inv_cache[x]

def solve():
    t = int(input())
    out = []
    for _ in range(t):
        m = int(input())
        primes = factorize(m)

        res = pow(m, 4, MOD)

        for p in primes:
            res = res * (p - 1) % MOD
            res = res * modinv(p) % MOD
            res = res * (p * p - 1) % MOD
            res = res * modinv(p * p) % MOD

        out.append(str(res % MOD))

    print("\n".join(out))

if __name__ == "__main__":
    solve()
