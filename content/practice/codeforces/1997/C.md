すべての有効な修復物の中で、可能な限り最小限のコストを見つけなければなりません。

import sys
input = sys.stdin.readline

def solve():
    t = int(input())

    for _ in range(t):
        n = int(input())
        s = input().strip()

        balance = 0
        cost = 0

        for i in range(1, n + 1):
            if i % 2 == 1:
                if balance == 0:
                    ch = '('
                else:
                    ch = ')'
            else:
                ch = s[i - 1]

            if ch == '(':

                cost -= i
            else:
                balance -= 1
                cost += i

        print(cost)
