## Python ソリューション```python
import sys
input = sys.stdin.readline

def egcd(a, b):
    if b == 0:
        return a, 1, 0
    g, x, y = egcd(b, a % b)
    return g, y, x - (a // b) * y

def mod_pow(a, e, mod):
    return pow(a, e, mod)

def solve():
    Q = int(input())
    out = []

    for _ in range(Q):
        p, a, b, x, y = map(int, input().split())

        x %= p
        y %= p

        # handle zero cases
        if x == 0 or y == 0:
            z = 0
            if pow(z, a, p) == y and pow(z, b, p) == x:
                out.append(str(p))
            else:
                out.append("-1")
            continue

        # Bézout: a*u + b*v = 1
        g, u, v = egcd(a, b)

        # g must be 1 since gcd(a,b)=1
        if g != 1:
            out.append("-1")
            continue

        # z = y^u * x^v mod p
        zu = pow(y, u, p) if u >= 0 else pow(pow(y, -u, p), p - 2, p)
        zv = pow(x, v, p) if v >= 0 else pow(pow(x, -v, p), p - 2, p)

        z = (zu * zv) % p

        if z == 0:
            z = p

        out.append(str(z))

    print("\n".join(out))

if __name__ == "__main__":
    solve()

最後の正規化ステップでは、モジュラー代表であるため、出力が正の整数であることが保証されます。$0$に対応する$p$必要な出力ドメイン内。
