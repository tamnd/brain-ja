## Python ソリューション```python
import sys
input = sys.stdin.readline

def can_reach(A, B):
    n = len(A)
    if A == B:
        return 0

    cur = B[:]
    steps = 0

    while True:
        if cur == A:
            return steps

        if len(cur) == 1:
            break

        nxt = [cur[i] - cur[i - 1] for i in range(1, n)]
        cur = nxt
        steps += 1

    return None

n = int(input())
A = list(map(int, input().split()))
B = list(map(int, input().split()))

res = can_reach(A, B)
res_rev = can_reach(A[::-1], B)

best = None
rev_used = False

if res is not None:
    best = res
if res_rev is not None:
    if best is None or res_rev < best:
        best = res_rev
        rev_used = True

if best is None:
    print("IMPOSSIBLE")
    sys.exit()

if best > 200000:
    print("BIG")
    print(best)
    sys.exit()

print("SMALL")
print(best)

ops = []
if rev_used:
    ops.append('R')

ops.extend(['P'] * best)

print(len(ops))
print(''.join(ops))
