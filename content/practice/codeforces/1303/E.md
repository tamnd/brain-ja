## Python ソリューション```python
import sys
input = sys.stdin.readline

def can_build(s, t):
    n, m = len(s), len(t)
    i = 0
    used = 1

    while used <= 2 and i < m:
        progress = False

        for ch in s:
            if i < m and ch == t[i]:
                i += 1
                progress = True

        if i == m:
            return True

        if not progress:
            return False

        used += 1

    return i == m

T = int(input())
for _ in range(T):
    s = input().strip()
    t = input().strip()
    print("YES" if can_build(s, t) else "NO")
