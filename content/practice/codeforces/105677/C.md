## Python ソリューション```python
import sys
input = sys.stdin.readline

INF = 10**9

def solve():
    n, m = map(int, input().split())
    s = input().strip()
    t = input().strip()

    dp = [[INF] * (m + 1) for _ in range(n + 1)]
    dp[0][0] = 0

    for i in range(n):
        ndp = [[INF] * (m + 1) for _ in range(n + 1)]

        for j in range(m + 1):
            if dp[i][j] == INF:
                continue

            cur = dp[i][j]

            if j < m:
                if s[i] == t[j]:
                    if cur + 1 < ndp[i + 1][j + 1]:
                        ndp[i + 1][j + 1] = cur + 1

            if cur < ndp[i + 1][j]:
                ndp[i + 1][j] = cur

            if j < m:
                if cur < ndp[i][j + 1]:
                    ndp[i][j + 1] = cur

        dp = ndp

    ans = min(dp[n])
    print(ans)

if __name__ == "__main__":
    solve()
