## Python ソリューション```python
import sys
from math import gcd

input = sys.stdin.readline

MOD = 1000000007
MAXN = 200000 + 5

fact = [1] * MAXN
for i in range(1, MAXN):
    fact[i] = fact[i - 1] * i % MOD

def is_power_of_two(x):
    return x > 0 and (x & (x - 1)) == 0

t = int(input())

for _ in range(t):
    n, c = map(int, input().split())
    s = input().strip()

    if c == 1:
        print(-1)
        continue

    if s[0] == '0' or s[-1] == '0':
        print(-1)
        continue

    ans = 1
    rem = c

    odd_candidates = []

    for i in range(1, n):
        ch = s[i - 1]

        if i == n:
            continue

        if ch == '1':
            ans = ans * 2 % MOD
            rem //= gcd(rem, 2)

        elif ch == '0':
            val = i - 1
            ans = ans * val % MOD
            rem //= gcd(rem, val)

        else:
            if i == 2:
                pass
            elif i >= 3:
                odd_candidates.append(i - 1)

    if rem == 1:
        print(-1)
        continue

    if not is_power_of_two(rem):
        for _ in odd_candidates:
            ans = ans * 2 % MOD
        print(ans)
        continue

    need = rem.bit_length() - 1

    m = len(odd_candidates)

    if m < need:
        for _ in odd_candidates:
            ans = ans * 2 % MOD
        print(ans)
        continue

    use_twos = need - 1

    odd_candidates.sort()

    for idx, j in enumerate(odd_candidates):
        if idx < use_twos:
            ans = ans * 2 % MOD
        else:
            ans = ans * j % MOD

    print(ans)
```最初のフェーズでは、すべての固定位置の寄与を計算します。 同時に、共通要素を削除します。$c$、還元された価値を生み出す$c'$。 

リスト`odd_candidates`すべての未解決の因子ペアを保存します$\{2,j\}$と$j=i-1\ge2$。 

支店`rem`が 2 のべき乗ではないのは簡単なケースです。 どこでも 2 を選択すると最小限であり、自動的に割り算が回避されます。 

いつ`rem`は 2 のべき乗であり、2 の指数のみが問題となります。 変数`need`必要な指数を格納します。 正確に守ります`need - 1`因数が 2 に等しく、残りを奇数の代替案に置き換えて、2 進数の評価が割り切れるしきい値を下回る最小の積を生成します。
