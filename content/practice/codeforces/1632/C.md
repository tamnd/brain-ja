## Python ソリューション```python
import sys
input = sys.stdin.readline

def solve():
    t = int(input())
    for _ in range(t):
        a, b = map(int, input().split())

        ans = b - a  # only increment a up to b

        # try a few possible "final alignment points"
        # it is enough to try b + k for small k
        for add in range(0, 33):
            x = b + add

            # cost to raise b to x
            cost_b = add

            # simulate best possible growth of a using OR
            cur_a = a
            cur_b = b + add
            ops = cost_b

            # greedily apply OR when it helps
            while cur_a < x:
                if (cur_a | cur_b) > cur_a:
                    cur_a = cur_a | cur_b
                    ops += 1
                else:
                    cur_a += 1
                    ops += 1

                if ops >= ans:
                    break

            ans = min(ans, ops)

        print(ans)

if __name__ == "__main__":
    solve()

## テストケース```python
import sys, io

def run(inp: str) -> str:
    sys.stdin = io.StringIO(inp)
    input = sys.stdin.readline

    t = int(input())
    out = []
    for _ in range(t):
        a, b = map(int, input().split())
        ans = b - a
        for add in range(0, 33):
            x = b + add
            cur_a = a
            cur_b = b + add
            ops = add
            while cur_a < x:
                if (cur_a | cur_b) > cur_a:
                    cur_a = cur_a | cur_b
                    ops += 1
                else:
                    cur_a += 1
                    ops += 1
                if ops >= ans:
                    break
            ans = min(ans, ops)
        out.append(str(ans))
    return "\n".join(out)
