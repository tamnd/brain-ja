## Python ソリューション```python
import sys
input = sys.stdin.readline

MOD = 998244353

sys.setrecursionlimit(10**7)

n = int(input())
a = list(map(int, input().split()))
b = list(map(int, input().split()))

left = [0] * (2 * n)
right = [0] * (2 * n)
children = [[] for _ in range(2 * n)]
parent = [-1] * (2 * n)

for i in range(1, n):
    x, y = map(int, input().split())
    left[i] = x
    right[i] = y
    children[i].append(x)
    children[i].append(y)
    parent[x] = i
    parent[y] = i

root = 1
while parent[root] != -1:
    root = parent[root]

L = [0] * (2 * n)
R = [0] * (2 * n)

def dfs(u):
    if u >= n:
        idx = u - n
        L[u] = R[u] = idx
        return L[u], R[u]
    l, r = dfs(left[u])
    dfs(right[u])
    L[u] = l
    R[u] = r
    return L[u], R[u]

dfs(root)

cnt = [0] * n

def add(u, val):
    if u == -1:
        return
    if u >= n:
        cnt[u - n] += val
        return
    add(left[u], val)
    add(right[u], val)

add(root, 1)

# initial answer
def build(u):
    if u >= n:
        return a[u - n]
    return build(left[u]) * build(right[u]) % MOD

ans = 0
def sumtree(u):
    global ans
    if u >= n:
        v = a[u - n]
        ans = (ans + v) % MOD
        return v
    v = sumtree(left[u]) * sumtree(right[u]) % MOD
    ans = (ans + v) % MOD
    return v

sumtree(root)

for i in range(n):
    ans = ans * pow(b[i], cnt[i], MOD) % MOD
    print(ans, end=' ')

重要な実装の詳細は、リーフがノードとして格納されるため、リーフのインデックス付けが入力インデックス付けと確実に一致するようにすることです。$n$に$2n-1$。 ここで不一致があると、間の対応関係が壊れます。$a_i$とその葉ノード。

 ## 特殊なケース

 縮退したチェーン形状のツリーでは、すべてのリーフ更新が概念的に長い依存関係パスに影響を与えます。 単純な伝播アプローチは更新するでしょう$O(n)$操作ごとのノード数。 事前に計算された$cnt[i]$トラバースを完全に回避します。 チェーンの下部にあるリーフの場合、指数は完全な深さを一度反映し、再利用されます。
