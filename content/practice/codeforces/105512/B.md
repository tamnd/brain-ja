## Python ソリューション```python
import sys
input = sys.stdin.readline

def solve():
    n, k = map(int, input().split())
    a = list(input().strip())
    b = list(input().strip())

    if a.count('1') != b.count('1'):
        print(-1)
        return

    if k == 0:
        if a == b:
            print(0)
        else:
            print(-1)
        return

    if k == a.count('1'):
        if a == b:
            print(0)
        else:
            print(-1)
        return

    ops = []

    def ones(l, r):
        return sum(1 for i in range(l, r+1) if a[i] == '1')

    for i in range(n):
        if a[i] == b[i]:
            continue

        found = False

        for r in range(i, n):
            for l in range(i, r+1):
                if ones(l, r) == k:
                    a[l:r+1] = a[l:r+1][::-1]
                    ops.append((l+1, r+1))
                    found = True
                    break
            if found:
                break

        if not found:
            print(-1)
            return

    print(len(ops))
    for l, r in ops:
        print(l, r)

def main():
    t = int(input())
    for _ in range(t):
        solve()

if __name__ == "__main__":
    main()

正確さは、必要なローカル調整が、正確に含まれる有効なセグメントによって実現できるという事実に依存します。$k$グローバルな変革が可能であれば。 実装ではロジックをシンプルに保ち、解決策が存在するという問題の保証に依存します。$4n$動きます。

 ## 特殊なケース

 いつ$k = 0$、すべての有効なセグメントには 1 が含まれていてはなりません。つまり、すべてゼロの部分文字列のみを反転できます。 このような反転によって文字列は変更されないため、文字列間の不一致は発生しません。$a$そして$b$すぐに失敗につながります。 アルゴリズムは、操作を試行する前にこれを捕捉します。
