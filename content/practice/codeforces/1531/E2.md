## Python ソリューション```python
import sys
input = sys.stdin.readline

s = input().strip()
ptr = 0
next_val = 1

def build(n):
    global ptr, next_val
    if n == 1:
        val = next_val
        next_val += 1
        return [val]
    m = n // 2
    left = build(m)
    right = build(n - m)
    merged = []
    i = j = 0
    while i < len(left) and j < len(right):
        if s[ptr] == '0':
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1
        ptr += 1
    merged.extend(left[i:])
    merged.extend(right[j:])
    return merged

# find minimal n such that n-1 <= len(s)
n = 1
while n*(n-1)//2 < len(s):
    n += 1

a = build(n)
print(n)
print(' '.join(map(str, a)))
