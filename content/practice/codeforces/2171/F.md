## Python ソリューション```python
import sys
input = sys.stdin.readline

def solve():
    t = int(input())
    for _ in range(t):
        n = int(input())
        p = list(map(int, input().split()))
        pos = [0]*(n+1)
        for i, x in enumerate(p):
            pos[x] = i
        
        if p[0] != 1:
            print("No")
            continue
        
        edges = []
        stack = [p[0]]
        
        possible = True
        for v in p[1:]:
            while stack and stack[-1] > v:
                stack.pop()
            if not stack:
                possible = False
                break
            edges.append((stack[-1], v))
            stack.append(v)
        
        if possible:
            print("Yes")
            for u, v in edges:
                print(u, v)
        else:
            print("No")

if __name__ == "__main__":
    solve()
