## Python ソリューション```python
import sys
input = sys.stdin.readline

def ask(v: int) -> int:
    print(2, v)
    sys.stdout.flush()
    return int(input())

def solve():
    n = int(input())
    for _ in range(n - 1):
        input()

    v = 1
    while True:
        nxt = ask(v)
        if nxt == v:
            print("! ", v, sep="")
            sys.stdout.flush()
            return
        v = nxt

if __name__ == "__main__":
    solve()
