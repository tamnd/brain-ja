## Python ソリューション```python
import sys
input = sys.stdin.readline

def phi(n: int) -> int:
    result = n
    p = 2
    while p * p <= n:
        if n % p == 0:
            while n % p == 0:
                n //= p
            result -= result // p
        p += 1
    if n > 1:
        result -= result // n
    return result

def solve():
    t = int(input())
    for _ in range(t):
        a, m = map(int, input().split())
        g = 0
        import math
        g = math.gcd(a, m)

        m_prime = m // g
        print(g * phi(m_prime))

if __name__ == "__main__":
    solve()
