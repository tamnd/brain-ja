[プロジェクト オイラー問題 972](https://projecteuler.net/problem=972)

 ## 解決策

 **答え:**`3575508`させて$$s(P)=x^2+y^2+1.$$ポアンカレ円板の双曲測地線は次のいずれかです。 

- 原点を通る直径、または
 - 単位円に直交する円。 

このような円には方程式があります$$x^2+y^2+Ax+By+1=0.$$そこで3つのポイント$P_i=(x_i,y_i)$共通の測地線が存在する場合、その上に位置する$A,B$そのような$$x_i^2+y_i^2+A x_i+B y_i+1=0$$みんなのために$i$。 排除する$A,B$決定条件を与えます$$\det \begin{pmatrix} x_1 & y_1 & x_1^2+y_1^2+1\\ x_2 & y_2 & x_2^2+y_2^2+1\\ x_3 & y_3 & x_3^2+y_3^2+1 \end{pmatrix}=0.$$したがって、すべての点は射影点に対応します$$(x,y,x^2+y^2+1),$$測地線は投影線に正確に対応します。 

計算戦略は次のとおりです。 

1. 分母を持つ単位ディスク内のすべての有理点を生成します。$\le 12$。 
2. 各点を整数射影ベクトルに変換します。 
3. すべてのペアが一意の射影線を決定します (外積によって)。 
4. 各線上にある点の数を数えます。 
5.合計$k(k-1)(k-2)$順序付けられたトリプルが必要であるため、すべての行にわたって実行されます。 

次の C++ プログラムは正確な計算を実行します。```python
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct Key {
    ll a,b,c;
};

struct H {
    size_t operator()(Key const& k) const noexcept {
        return ((uint64_t)k.a * 1315423911u)
             ^ ((uint64_t)k.b << 21)
             ^ k.c;
    }
};

struct E {
    bool operator()(Key const& x, Key const& y) const noexcept {
        return x.a==y.a && x.b==y.b && x.c==y.c;
    }
};

ll gcd3(ll a,ll b,ll c){
    return gcd(gcd(llabs(a), llabs(b)), llabs(c));
}

int main() {
    int N = 12;

    vector<pair<int,int>> vals;
    set<pair<int,int>> seen;

    // all reduced rationals with denominator <= N
    for(int q=1;q<=N;q++){
        for(int p=-q+1;p<q;p++){
            int g = gcd(abs(p), q);
            pair<int,int> t = {p/g, q/g};
            if(!seen.count(t)){
                seen.insert(t);
                vals.push_back(t);
            }
        }
    }

    // projective points
    vector<array<int,3>> pts;

    for(auto [p,q] : vals){
        for(auto [r,s] : vals){

            if(1LL*p*p*s*s + 1LL*r*r*q*q < 1LL*q*q*s*s){

                int g = gcd(q,s);
                int l = q/g*s;

                ll X = 1LL*p*(l/q);
                ll Y = 1LL*r*(l/s);

                ll a = X*l;
                ll b = Y*l;
                ll c = X*X + Y*Y + 1LL*l*l;

                ll gg = gcd3(a,b,c);

                pts.push_back({
                    (int)(a/gg),
                    (int)(b/gg),
                    (int)(c/gg)
                });
            }
        }
    }

    unordered_map<Key,int,H,E> mp;
    mp.reserve(30000000);

    int n = pts.size();

    // each pair determines a line
    for(int i=0;i<n;i++){
        auto [a1,b1,c1] = pts[i];

        for(int j=i+1;j<n;j++){

            auto [a2,b2,c2] = pts[j];

            ll A = 1LL*b1*c2 - 1LL*c1*b2;
            ll B = 1LL*c1*a2 - 1LL*a1*c2;
            ll C = 1LL*a1*b2 - 1LL*b1*a2;

            ll g = gcd3(A,B,C);

            A/=g; B/=g; C/=g;

            if(A<0 || (A==0 && (B<0 || (B==0 && C<0)))){
                A=-A; B=-B; C=-C;
            }

            mp[{A,B,C}]++;
        }
    }

    long long ans = 0;

    // if a line has k points, it contributes k(k-1)(k-2)
    for(auto &kv : mp){

        long long v = kv.second;

        // v = C(k,2)
        long long s = sqrtl(1 + 8*v);
        long long k = (1 + s)/2;

        ans += k*(k-1)*(k-2);
    }

    cout << ans << endl;
}
